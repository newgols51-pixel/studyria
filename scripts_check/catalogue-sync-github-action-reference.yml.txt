name: Catalogue Crawler Snapshot Sync

# Auto-syncs the crawler-readable static catalogue (CATALOGUE-SNAPSHOT block,
# stat placeholders, JSON-LD ItemList, <noscript> catalogue) in index.html
# with the LIVE production Supabase catalogue.
#
# Flow:  Supabase anon query → sync-static-catalogue.mjs → regenerate →
#        commit only if changed → push to main → Cloudflare git integration
#        auto-deploys (same mechanism as build-pages.yml).
#
# Failure safety: the script exits 1 (and writes NOTHING) on any fetch
# failure or zero-row result — the last known-good catalogue is preserved
# and this workflow reports the failure without pushing.

on:
  schedule:
    # Daily at 03:30 UTC (09:00 IST) — after the 02:00 build-pages run, so
    # new PDFs published via Smart Publish appear in the crawler catalogue
    # within ~90 minutes at most.
    - cron: '30 3 * * *'
  workflow_dispatch: {}

permissions:
  contents: write

concurrency:
  group: catalogue-sync
  cancel-in-progress: false

jobs:
  sync-catalogue:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: Sync crawler catalogue from production DB
        run: node scripts_check/sync-static-catalogue.mjs
        env:
          # Public anon key (same one shipped in the client bundle).
          # Reuses the repo secrets already configured for build-pages.yml.
          SUPABASE_URL: ${{ secrets.SUPABASE_URL }}
          SUPABASE_ANON_KEY: ${{ secrets.SUPABASE_ANON }}

      - name: Commit + push regenerated catalogue (only when changed)
        run: |
          git config user.name "studyria-bot"
          git config user.email "actions@users.noreply.github.com"
          git add index.html sw.js scripts_check/catalogue-last-sync.json
          git diff --cached --quiet && echo "Catalogue unchanged — nothing to deploy." && exit 0
          git commit -m "chore: auto-sync crawler catalogue from production DB [skip ci]"
          git pull --rebase origin main
          git push origin main
          echo "PUSHED — Cloudflare auto-deploys the new catalogue."

      - name: Report sync summary
        if: always()
        run: |
          echo "── Catalogue sync run ──"
          echo "Timestamp: $(date -u +%Y-%m-%dT%H:%M:%SZ)"
          if [ -f scripts_check/catalogue-last-sync.json ]; then
            node -e "const s=require('./scripts_check/catalogue-last-sync.json'); console.log('Last synced state: '+s.pdfCount+' PDFs, '+s.categoryCount+' categories, '+s.downloadAggregate+' downloads @ '+s.syncedAt)"
          else
            echo "No sync state file present."
          fi
